
import os
import logging
from pathlib import Path

import psycopg2
from psycopg2.extras import RealDictCursor
from dotenv import load_dotenv


# ==========================================
# PATHS AND ENVIRONMENT VARIABLES
# ==========================================

BASE_DIR = Path(__file__).resolve().parent.parent
BACKEND_DIR = Path(__file__).resolve().parent

# Load environment variables from project root and backend folder.
# Existing system environment variables are not overwritten.
load_dotenv(BASE_DIR / ".env")
load_dotenv(BACKEND_DIR / ".env")

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


# ==========================================
# DATABASE CONNECTION
# ==========================================

def get_connection():
    """Create a PostgreSQL connection using DATABASE_URL."""

    database_url = (
        os.getenv("DATABASE_URL")
        or os.getenv("POSTGRES_URL")
        or os.getenv("POSTGRES_PRISMA_URL")
    )

    if not database_url or not database_url.strip():
        raise RuntimeError(
            "DATABASE_URL is missing. "
            "Add your PostgreSQL connection URL to the .env file "
            "or configure DATABASE_URL in your deployment environment."
        )

    database_url = database_url.strip()

    # Support older PostgreSQL URL prefixes.
    if database_url.startswith("postgres://"):
        database_url = "postgresql://" + database_url[len("postgres://"):]

    return psycopg2.connect(
        database_url,
        cursor_factory=RealDictCursor,
        connect_timeout=15,
        sslmode="require",
    )


# ==========================================
# CREATE DATABASE TABLES
# ==========================================

def create_history_table():
    """Create required tables when a database is configured."""

    if not (
        os.getenv("DATABASE_URL")
        or os.getenv("POSTGRES_URL")
        or os.getenv("POSTGRES_PRISMA_URL")
    ):
        logger.warning(
            "DATABASE_URL is not configured. "
            "Database tables were not initialized."
        )
        return False

    connection = None

    try:
        connection = get_connection()

        with connection.cursor() as cursor:
            cursor.execute("""
                CREATE TABLE IF NOT EXISTS prediction_history (
                    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
                    song TEXT NOT NULL,
                    mood TEXT NOT NULL,
                    confidence DOUBLE PRECISION NOT NULL DEFAULT 0,
                    intensity TEXT NOT NULL DEFAULT 'Low',
                    date TEXT NOT NULL
                )
            """)

            cursor.execute("""
                CREATE TABLE IF NOT EXISTS user_feedback (
                    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
                    song_name TEXT NOT NULL DEFAULT '',
                    mood TEXT NOT NULL DEFAULT '',
                    confidence DOUBLE PRECISION NOT NULL DEFAULT 0,
                    helpfulness TEXT NOT NULL DEFAULT '',
                    rating INTEGER CHECK (rating IS NULL OR rating BETWEEN 1 AND 5),
                    feedback TEXT NOT NULL DEFAULT '',
                    submitted_at TEXT NOT NULL DEFAULT ''
                )
            """)

        connection.commit()
        logger.info("PostgreSQL tables initialized successfully.")
        return True

    except Exception:
        if connection is not None:
            connection.rollback()
        logger.exception("Failed to initialize PostgreSQL tables.")
        raise

    finally:
        if connection is not None:
            connection.close()


# ==========================================
# SAVE PREDICTION
# ==========================================

def save_prediction(song, mood, confidence, intensity, date):
    connection = None

    try:
        connection = get_connection()

        with connection.cursor() as cursor:
            cursor.execute(
                """
                INSERT INTO prediction_history
                    (song, mood, confidence, intensity, date)
                VALUES (%s, %s, %s, %s, %s)
                RETURNING id
                """,
                (
                    str(song),
                    str(mood),
                    float(confidence),
                    str(intensity),
                    str(date),
                ),
            )
            result = cursor.fetchone()

        connection.commit()
        return dict(result) if result else None

    except Exception:
        if connection is not None:
            connection.rollback()
        logger.exception("Failed to save prediction.")
        raise

    finally:
        if connection is not None:
            connection.close()


# ==========================================
# GET PREDICTION HISTORY
# ==========================================

def get_predictions():
    connection = None

    try:
        connection = get_connection()

        with connection.cursor() as cursor:
            cursor.execute("""
                SELECT id, song, mood, confidence, intensity, date
                FROM prediction_history
                ORDER BY id DESC
            """)
            rows = cursor.fetchall()

        return [dict(row) for row in rows]

    except Exception:
        logger.exception("Failed to fetch prediction history.")
        raise

    finally:
        if connection is not None:
            connection.close()


# ==========================================
# DELETE ONE PREDICTION
# ==========================================

def delete_prediction(prediction_id):
    connection = None

    try:
        connection = get_connection()

        with connection.cursor() as cursor:
            cursor.execute(
                "DELETE FROM prediction_history WHERE id = %s",
                (int(prediction_id),),
            )
            deleted = cursor.rowcount > 0

        connection.commit()
        return deleted

    except Exception:
        if connection is not None:
            connection.rollback()
        logger.exception("Failed to delete prediction.")
        raise

    finally:
        if connection is not None:
            connection.close()


# ==========================================
# CLEAR ALL PREDICTIONS
# ==========================================

def clear_predictions():
    connection = None

    try:
        connection = get_connection()

        with connection.cursor() as cursor:
            cursor.execute("DELETE FROM prediction_history")
            deleted_count = cursor.rowcount

        connection.commit()
        return deleted_count

    except Exception:
        if connection is not None:
            connection.rollback()
        logger.exception("Failed to clear prediction history.")
        raise

    finally:
        if connection is not None:
            connection.close()


# ==========================================
# SAVE USER FEEDBACK
# ==========================================

def save_feedback(feedback_data):
    if not isinstance(feedback_data, dict):
        raise ValueError("Feedback must be provided as a dictionary.")

    song_name = str(
        feedback_data.get("songName")
        or feedback_data.get("song_name")
        or ""
    ).strip()

    mood = str(feedback_data.get("mood") or "").strip()
    helpfulness = str(feedback_data.get("helpfulness") or "").strip()
    feedback = str(feedback_data.get("feedback") or "").strip()
    submitted_at = str(
        feedback_data.get("submittedAt")
        or feedback_data.get("submitted_at")
        or ""
    ).strip()

    try:
        confidence = float(feedback_data.get("confidence") or 0)
    except (TypeError, ValueError):
        raise ValueError("Confidence must be a valid number.")

    rating_value = feedback_data.get("rating")
    rating = None

    if rating_value not in (None, ""):
        try:
            rating = int(rating_value)
        except (TypeError, ValueError):
            raise ValueError("Rating must be an integer from 1 to 5.")

        if rating < 1 or rating > 5:
            raise ValueError("Rating must be between 1 and 5.")

    connection = None

    try:
        connection = get_connection()

        with connection.cursor() as cursor:
            cursor.execute(
                """
                INSERT INTO user_feedback
                    (
                        song_name,
                        mood,
                        confidence,
                        helpfulness,
                        rating,
                        feedback,
                        submitted_at
                    )
                VALUES (%s, %s, %s, %s, %s, %s, %s)
                RETURNING id
                """,
                (
                    song_name,
                    mood,
                    confidence,
                    helpfulness,
                    rating,
                    feedback,
                    submitted_at,
                ),
            )
            result = cursor.fetchone()

        connection.commit()
        return dict(result) if result else None

    except Exception:
        if connection is not None:
            connection.rollback()
        logger.exception("Failed to save user feedback.")
        raise

    finally:
        if connection is not None:
            connection.close()
